- "ad0":  "Adult (Normal)",
- "ad1":  "Adult (Tatkal)",
- "ch0":  "Child (Normal)",
- "ch1":  "Child (Tatkal)",
- "srf0": "Senior Female (Normal)",
- "srf1": "Senior Female (Tatkal)",
- "srm0": "Senior Male (Normal)",
- "srm1": "Senior Male (Tatkal)"

    - 1 suffix = Tatkal fare
